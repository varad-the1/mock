-- PS10: Vehicle Fleet Maintenance (MySQL) - cursor + triggers
CREATE DATABASE IF NOT EXISTS fleet; USE fleet;
CREATE TABLE Vehicle(VehID INT PRIMARY KEY, RegNo VARCHAR(15), LastService DATE);
CREATE TABLE Maintenance_Stage(StageID INT PRIMARY KEY, VehID INT, ServiceDate DATE, Cost DECIMAL(8,2));
CREATE TABLE Maintenance_Master(StageID INT PRIMARY KEY, VehID INT, ServiceDate DATE, Cost DECIMAL(8,2));
CREATE TABLE Maintenance_Audit(AuditID INT AUTO_INCREMENT PRIMARY KEY, StageID INT, Action VARCHAR(10), NewCost DECIMAL(8,2), At TIMESTAMP DEFAULT CURRENT_TIMESTAMP);

DELIMITER //
CREATE TRIGGER trg_m_ins AFTER INSERT ON Maintenance_Master FOR EACH ROW
  INSERT INTO Maintenance_Audit(StageID,Action,NewCost) VALUES(NEW.StageID,'INSERT',NEW.Cost)//
CREATE TRIGGER trg_m_upd AFTER UPDATE ON Maintenance_Master FOR EACH ROW
  INSERT INTO Maintenance_Audit(StageID,Action,NewCost) VALUES(NEW.StageID,'UPDATE',NEW.Cost)//

CREATE PROCEDURE process_maintenance()
BEGIN
  DECLARE done INT DEFAULT 0; DECLARE s,v INT; DECLARE d DATE; DECLARE c DECIMAL(8,2);
  DECLARE cur CURSOR FOR SELECT StageID,VehID,ServiceDate,Cost FROM Maintenance_Stage;
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done=1;
  OPEN cur;
  lp: LOOP
    FETCH cur INTO s,v,d,c;
    IF done=1 THEN LEAVE lp; END IF;
    IF c>=0 AND EXISTS(SELECT 1 FROM Vehicle WHERE VehID=v)
       AND NOT EXISTS(SELECT 1 FROM Maintenance_Master WHERE StageID=s) THEN
      INSERT INTO Maintenance_Master VALUES(s,v,d,c);
      UPDATE Vehicle SET LastService=d WHERE VehID=v AND d>LastService;
    END IF;
  END LOOP;
  CLOSE cur;
END//
DELIMITER ;
