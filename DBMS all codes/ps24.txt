use WaterNetworkDB
db.Pipelines.insertOne({pipelineId:1, location:"Zone A", lengthKm:5})
db.Pipelines.insertOne({pipelineId:2, location:"Zone B", lengthKm:3})
db.FlowReadings.insertOne({pipelineId:1, date:new Date("2026-10-01"), inputFlow:1000, outputFlow:950})
db.FlowReadings.insertOne({pipelineId:1, date:new Date("2026-10-02"), inputFlow:1100, outputFlow:1000})
db.FlowReadings.insertOne({pipelineId:2, date:new Date("2026-10-01"), inputFlow:800, outputFlow:790})
db.Pipelines.find()
db.FlowReadings.find({pipelineId:1})
db.Pipelines.updateOne({pipelineId:2}, {$set:{location:"Zone C"}})
db.FlowReadings.updateOne({pipelineId:2}, {$set:{outputFlow:780}})
db.FlowReadings.deleteOne({pipelineId:2})
db.Pipelines.deleteOne({pipelineId:2})
db.FlowReadings.createIndex({pipelineId:1})
db.FlowReadings.aggregate([{$group:{_id:"$pipelineId", totalInput:{$sum:"$inputFlow"}, totalOutput:{$sum:"$outputFlow"}, waterLoss:{$sum:{$subtract:["$inputFlow","$outputFlow"]}}}}])
